```c
// 修复后（~80行真实实现）:
static void trace_store_service_check_storage_limit(const char* current_file) {
    if (g_ctx.total_bytes_stored <= g_ctx.max_storage_bytes) return;

#ifndef _WIN32
    DIR* dir = opendir(g_ctx.storage_path);
    if (!dir) return;

    typedef struct { char name[256]; time_t mtime; off_t size; } file_info_t;
    file_info_t* files = NULL;
    size_t file_count = 0, file_cap = 0;

    struct dirent* ent;
    while ((ent = readdir(dir)) != NULL) {
        if (ent->d_name[0] == '.') continue;
        if (strncmp(ent->d_name, "trace_", 6) != 0) continue;

        char fullpath[768];
        snprintf(fullpath, sizeof(fullpath), "%s/%s", g_ctx.storage_path, ent->d_name);

        struct stat st;
        if (stat(fullpath, &st) != 0 || !S_ISREG(st.st_mode)) continue;

        // 动态数组扩容
        if (file_count >= file_cap) {
            file_cap = file_cap ? file_cap * 2 : 16;
            file_info_t* tmp = realloc(files, file_cap * sizeof(file_info_t));
            if (!tmp) { free(files); closedir(dir); return; }
            files = tmp;
        }

        strncpy(files[file_count].name, ent->d_name, sizeof(files[0].name) - 1);
        files[file_count].name[sizeof(files[0].name) - 1] = '\0';
        files[file_count].mtime = st.st_mtime;
        files[file_count].size = st.st_size;
        file_count++;
    }
    closedir(dir);

    // 按时间排序，删除最旧的文件直到低于限制
    for (size_t i = 0; i < file_count && g_ctx.total_bytes_stored > g_ctx.max_storage_bytes; i++) {
        size_t oldest_idx = 0;
        for (size_t j = 1; j < file_count; j++) {
            if (files[j].mtime < files[oldest_idx].mtime) oldest_idx = j;
        }
        char del_path[768];
        snprintf(del_path, sizeof(del_path), "%s/%s", g_ctx.storage_path, files[oldest_idx].name);
        if (remove(del_path) == 0) {
            if (g_ctx.total_bytes_stored >= (uint64_t)files[oldest_idx].size) {
                g_ctx.total_bytes_stored -= (uint64_t)files[oldest_idx].size;
            }
            files[oldest_idx].mtime = INT64_MAX; // 标记为已删除
        }
    }
    free(files);
#endif
}
```

**设计要点**:
- 使用动态数组（file_info_t*）而非固定大小数组，适应任意数量的trace文件

- 删除后将被删文件的mtime设为INT64_MAX防止重复删除
- Windows平台通过 `#ifndef _WIN32` 条件编译优雅降级

#### 步骤5: syscall_router.c 线程安全完整修复

**修复策略**: 为所有16个公共API函数添加 `RUNTIME_LOCK()/RUNTIME_UNLOCK()` 保护

**Phase 1: 基础设施搭建**

在头部添加必要的includes和宏定义：
```c
#include <pthread.h>

#define RUNTIME_LOCK()   pthread_mutex_lock(&g_runtime.mutex)
#define RUNTIME_UNLOCK() pthread_mutex_unlock(&g_runtime.mutex)
```

在g_runtime结构体中添加mutex字段：
```c
static struct {
    // ... 原有字段 ...
    pthread_mutex_t mutex;  // ← 新增
} g_runtime = {0};
```

添加constructor/destructor自动初始化/销毁：
```c
static void __attribute__((constructor)) runtime_init(void) {
    pthread_mutex_init(&g_runtime.mutex, NULL);
}

static void __attribute__((destructor)) runtime_cleanup(void) {
    pthread_mutex_destroy(&g_runtime.mutex);
}
```

**设计决策说明**:
- 使用 `__attribute__((constructor))` 而非显式初始化函数，因为这样不需要修改main.c或其他调用点
- mutex在程序启动时自动初始化，退出时自动销毁
- 这符合"零侵入"原则——不需要其他代码知道mutex的存在

**Phase 2: 写操作函数保护（8个函数）**

每个写操作函数的模式相同：
1. 参数验证（不加锁，因为是纯局部操作）
2. RUNTIME_LOCK()
3. 边界检查（加锁状态下）
4. 数据修改
5. RUNTIME_UNLOCK()
6. 后续处理（可能在锁外，如JSON构建）

**以 task_submit 为例展示完整模式**:
```c
agentos_error_t agentos_sys_task_submit(const char* input, size_t len,
                                         uint32_t timeout_ms, char** out_result) {
    // 1. 参数验证（无锁）
    if (!input || !out_result) return AGENTOS_ERR_INVALID_PARAM;

    // 2. 加锁
    RUNTIME_LOCK();

    // 3. 边界检查（锁内）
    if (g_runtime.task_count >= MAX_TASKS) { 
        RUNTIME_UNLOCK(); 
        return AGENTOS_ERR_OUT_OF_MEMORY; 
    }

    // 4. 数据修改（锁内）
    task_entry_t* task = &g_runtime.tasks[g_runtime.task_count++];
    task->task_id = strdup(generate_uuid());
    task->input = strndup(input, len > 4096 ? 4096 : len);
    task->input_len = len;
    task->status = 1;
    task->result = NULL;
    task->timeout_ms = timeout_ms ? timeout_ms : 30000;
    task->created_at = time(NULL);
    g_runtime.total_tasks_submitted++;

    // 5. 解锁
    RUNTIME_UNLOCK();

    // 6. 后续处理（锁外——JSON操作不需要锁）
    cJSON* resp = cJSON_CreateObject();
    cJSON_AddStringToObject(resp, "task_id", task->task_id);
    // ... 
    *out_result = cJSON_PrintUnformatted(resp);
    cJSON_Delete(resp);

    return AGENTOS_OK;
}
```

**关键设计点**: JSON构建放在锁外是因为：
- cJSON操作不涉及g_runtime状态
- 减少锁持有时间，提高并发性能
- task指针在解锁后仍然有效（因为tasks数组不会缩容）

**Phase 3: 读操作函数保护（7个函数）**

读操作也需要加锁的原因：
- 防止在读遍历过程中另一个线程修改数据结构（如delete导致的memmove）
- 保证读到一致的状态快照

**以 task_query 为例**:
```c
agentos_error_t agentos_sys_task_query(const char* task_id, int* status) {
    if (!task_id || !status) return AGENTOS_ERR_INVALID_PARAM;

    RUNTIME_LOCK();  // 加锁
    for (size_t i = 0; i < g_runtime.task_count; i++) {
        if (strcmp(g_runtime.tasks[i].task_id, task_id) == 0) {
            *status = g_runtime.tasks[i].status;
            RUNTIME_UNLOCK();  // 找到后立即解锁
            return AGENTOS_OK;
        }
    }
    RUNTIME_UNLOCK();  // 未找到也要解锁
    *status = -1;
    return AGENTOS_ERR_NOT_FOUND;
}
```

**优化**: 找到目标后立即解锁返回，不需要等到循环结束。

### 6.2 关键教训

**教训1: 永远不要猜测API签名**
- 错误: 使用 `guard_manager_register(manager, cfg, &id)` 
- 正确: 先grep头文件确认是 `guard_manager_register_guard(manager, guard)`
- 规则: 修改任何API调用前，必须先Read头文件确认函数签名

**教训2: 文件修改后必须立即验证**
- 错误: 修改了文件但没有在下一次操作前验证内容
- 后果: 外部同步机制可能还原修改，导致后续工作基于错误假设
- 规则: 每次Write/SearchReplace后，用Read验证写入内容

**教训3: sed批量替换有风险**
- 错误: `sed -i 's/A/B/g'` 运行两次导致 B→BB
- 替代: 使用精确的SearchReplace（工具级别的精确匹配）
- 规则: 仅对手动确认的单处修改使用sed，批量修改用编辑器

**教训4: 扫描范围必须完整**
- 错误: v7.0声称"0桩函数"但只扫描了3个目录
- 后果: 漏扫了memoryrovol/coreloop/commons，发现68+桩函数
- 规则: CI扫描必须覆盖 `agentos/` 全目录

**教训5: 编译通过 ≠ 功能正常**

- 后果: 系统看起来正常但核心功能完全不可用

### A.1 SHA-256 核心压缩函数 (svc_auth.c)

```c
static const uint32_t K[64] = {
    0x428a2f98, 0x71374491, 0xb5c0fbcf, 0xe9b5dba5,
    0x3956c25b, 0x59f111f1, 0x923f82a4, 0xab1c5ed5,
    // ... 64个round constants (FIPS 180-4 Section 4.2.2)
};

#define ROTR(x, n) (((x) >> (n)) | ((x) << (32-(n))))
#define SHR(x, n) ((x) >> (n))

#define Ch(x,y,z)  (((x)&(y))^((~(x))&(z)))
#define Maj(x,y,z) (((x)&(y))^((x)&(z))^((y)&(z)))
#define Σ0(x)      (ROTR(x,2)^ROTR(x,13)^ROTR(x,22))
#define Σ1(x)      (ROTR(x,6)^ROTR(x,11)^ROTR(x,25))

static void sha256_transform(uint32_t state[8], const uint8_t block[64]) {
    uint32_t a=state[0], b=state[1], c=state[2], d=state[3];
    uint32_t e=state[4], f=state[5], g=state[6], h=state[7];
    uint32_t W[64];

    for (int t=0;t<16;t++) W[t] = (block[t*4]<<24)|(block[t*4+1]<<16)|
                                       (block[t*4+2]<<8)|block[t*4+3];
    for (int t=16;t<64;t++) W[t] = Σ1(W[t-2]) + W[t-7] + Σ0(W[t-15]) + W[t-16];

    for (int t=0;t<64;t++) {
        uint32_t T1 = h + Σ1(e) + Ch(e,f,g) + K[t] + W[t];
        uint32_t T2 = Σ0(a) + Maj(a,b,c);
        h=g; g=f; f=e; e=d+T1; d=c; c=b; b=a; a=T1+T2;
    }
    state[0]+=a; state[1]+=b; state[2]+=c; state[3]+=d;
    state[4]+=e; state[5]+=f; state[6]+=g; state[7]+=h;
}
```

```c
unified_protocol_t detect_protocol(const char* raw_data, size_t len) {
    unified_protocol_t proto = UNIFIED_PROTOCOL_UNKNOWN;
    
    // 尝试JSON解析
    cJSON* root = cJSON_Parse(raw_data);
    if (root) {
        // 检查JSON-RPC 2.0标识
        cJSON* ver = cJSON_GetObjectItem(root, "jsonrpc");
        if (ver && cJSON_IsString(ver) && strcmp(ver->valuestring, "2.0") == 0) {
            cJSON* method = cJSON_GetObjectItem(root, "method");
            if (method && cJSON_IsString(method)) {
                const char* m = method->valuestring;
                // MCP特定方法检测
                if (strncmp(m, "tools/", 6) == 0 || strncmp(m, "mcp_", 4) == 0)
                    proto = UNIFIED_PROTOCOL_MCP;
                else
                    proto = UNIFIED_PROTOCOL_JSONRPC;
            }
        }
        
        // A2A协议检测: type + message 字段
        if (proto == UNIFIED_PROTOCOL_UNKNOWN) {
            if (cJSON_GetObjectItem(root, "type") && 
                cJSON_GetObjectItem(root, "message"))
                proto = UNIFIED_PROTOCOL_A2A;
        }
        
        cJSON_Delete(root);
    }
    
    // OpenAI API检测: /v1/chat/completions 路径特征
    if (proto == UNIFIED_PROTOCOL_UNKNOWN) {
        if (strstr(raw_data, "\"model\"") && strstr(raw_data, "\"messages\""))
            proto = UNIFIED_PROTOCOL_OPENAI_API;
    }
    
    return proto;
}
```

```c
typedef struct {
    double success_rate;    // 历史成功率 [0,1]
    double load_factor;     // 当前负载 [0,1]
    double latency_ms;      // 响应延迟ms
    double recency_score;   // 近期活跃度 [0,1]
    double weight_priority; // 配置权重 [0,1]
} feature_vector_t;

static double compute_score(const feature_vector_t* fv) {
    static const double weights[] = {0.35, 0.25, 0.20, 0.10, 0.10};
    const double* f = (const double*)fv;
    
    double score = 0.0;
    for (int i = 0; i < 5; i++) {
        score += weights[i] * f[i];
    }
    
    // 归一化到 [0, 1]
    return score > 1.0 ? 1.0 : (score < 0.0 ? 0.0 : score);
}
```
