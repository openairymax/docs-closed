# 推理语言全生命周期管理网关

## 1. 核心理念

用户输入的自然语言不应该直接"裸奔"进模型，必须经过网关的标准化处理，转变为**内部标准数据格式（Canonical Data Format）**后，再路由给具体模型。

---

# ======================== Phase 1: 决策前信号提取 ========================

class SignalExtractor:
    """轻量级信号提取器 (< 15ms 完成)"""
    
    # 文化/政策类关键词 (硬约束)
    CULTURE_KEYWORDS = ["成语", "典故", "古诗", "诗词", "论语", "道德经", "孙子兵法",
                        "四大名著", "红楼梦", "西游记", "三国演义", "水浒传"]
    POLICY_KEYWORDS = ["政策", "法规", "法律", "宪法", "民法典", "行政法规", "国务院"]
    
    # 代码类特征
    CODE_PATTERNS = [
        r'```\w*',           # markdown 代码块
        r'def\s+\w+\s*\(',   # python 函数
        r'function\s+\w+\s*\(', # js 函数
        r'class\s+\w+',      # 类定义
        r'import\s+\w+',     # import 语句
    ]
    
    @classmethod
    def detect_language(cls, text: str) -> Tuple[Language, float]:
        """语言检测, 带置信度"""
        if not text or len(text.strip()) < 3:
            return Language.UNKNOWN, 0.0
        
        # 1. 代码块检测 (避免把代码当自然语言)
        code_ratio = sum(1 for c in text if not c.isprintable() or c in '{}[]()=+*/-') / max(len(text), 1)
        if code_ratio > 0.35:
            # 可能包含大量代码, 标记为英文 (英文代码更省token)
            return Language.EN, 0.6
        
        # 2. 非ASCII字符比例 (粗略判断中英文)
        ascii_ratio = sum(1 for c in text if ord(c) < 128) / max(len(text), 1)
        if ascii_ratio > 0.85:
            return Language.EN, 0.9
        
        # 3. 使用 langdetect (生产环境建议用 fasttext-langdetect, 延迟<5ms)
        try:
            results = detect_langs(text[:500])  # 截断防止超长文本
            if results and results[0].prob >= 0.85:
                lang_code = results[0].lang.split('-')[0].lower()
                if lang_code in ['zh', 'zh-cn', 'zh-tw']:
                    return Language.ZH, results[0].prob
                elif lang_code in ['en', 'en-us', 'en-gb']:
                    return Language.EN, results[0].prob
        except Exception:
            pass
        
        # 4. 回退: 按中文字符占比
        chinese_ratio = sum(1 for c in text if '\u4e00' <= c <= '\u9fff') / max(len(text), 1)
        if chinese_ratio > 0.3:
            return Language.ZH, chinese_ratio
        return Language.EN, 0.5
    
    @classmethod
    def detect_task_type(cls, text: str) -> TaskType:
        """任务类型分类 (基于启发式)"""
        text_lower = text.lower()
        
        # 代码生成
        if any(p in text for p in ['写代码', '实现', 'function', 'def ', 'class ', 'import ']) or \
           any(re.search(p, text) for p in cls.CODE_PATTERNS):
            return TaskType.CODE
        
        # 数学推理
        if any(k in text for k in ['计算', '方程', '导数', '积分', '矩阵', '∑', '∫', 'solve', 'equation']):
            return TaskType.MATH
        
        # 文化类 (强制中文)
        if any(k in text for k in cls.CULTURE_KEYWORDS):
            return TaskType.CULTURE
        
        # 政策类 (强制中文)
        if any(k in text for k in cls.POLICY_KEYWORDS):
            return TaskType.POLICY
        
        return TaskType.QA
    
    @classmethod
    def extract(cls, raw_input: str, history: List[Dict]) -> Signals:
        """提取所有信号"""
        # 语言检测
        lang, conf = cls.detect_language(raw_input)
        
        # 任务类型
        task = cls.detect_task_type(raw_input)
        
        # 历史长度估算 (简单: 中文字数*1.8 + 英文词数*1.2)
        history_tokens = 0
        for turn in history:
            content = turn.get('content', '')
            chinese_chars = sum(1 for c in content if '\u4e00' <= c <= '\u9fff')
            english_words = len(re.findall(r'[a-zA-Z]+', content))
            history_tokens += int(chinese_chars * 1.8 + english_words * 1.2)
        
        # 代码含量判断
        code_indicators = sum(1 for p in cls.CODE_PATTERNS if re.search(p, raw_input))
        
        return Signals(
            detected_input_lang=lang,
            lang_confidence=conf,
            task_type=task,
            history_length_tokens=history_tokens,
            is_critical=task in [TaskType.CULTURE, TaskType.POLICY],
            contains_code=code_indicators > 0
        )


# ======================== Phase 1: 路由决策引擎 ========================

@dataclass
class ModelProfile:
    """模型画像 (从数据库加载)"""
    model_id: str
    provider: str          # openai | anthropic | deepseek | qwen | local
    family: str            # GPT | Claude | DeepSeek | Qwen | Llama
    native_lang: Language  # en | zh
    lang_ratio: float      # 中文token数 / 英文token数
    calibrated_at: str


class Router:
    """多因子路由决策引擎"""
    
    @classmethod
    def decide(cls, signals: Signals, model_profile: ModelProfile) -> RoutingDecision:
        decision = RoutingDecision(target_model=model_profile.model_id)
        chain = []
        
        # ---- 硬规则层 (不可逾越) ----
        if signals.task_type in [TaskType.CULTURE, TaskType.POLICY]:
            decision.reasoning_lang = Language.ZH
            decision.output_lang = Language.ZH
            decision.decision_reason = "hard_rule_culture_policy"
            chain.append("hard_rule: culture/policy → force_zh")
            decision.decision_chain = chain
            return decision
        
        # ---- 模型原生能力层 ----
        # Qwen/DeepSeek 在代码/数学任务上中文推理更优
        if signals.task_type in [TaskType.CODE, TaskType.MATH] and model_profile.family in ['Qwen', 'DeepSeek']:
            decision.reasoning_lang = Language.ZH
            decision.output_lang = signals.detected_input_lang  # 跟随用户输入
            decision.decision_reason = "native_advantage_code_math"
            chain.append("native: Qwen/DeepSeek + code/math → zh")
            decision.decision_chain = chain
            return decision
        
        # ---- 输入语言对齐层 ----
        # 若用户输入英文且模型英文效率高 → 英文推理
        if signals.detected_input_lang == Language.EN and model_profile.lang_ratio > 1.2:
            decision.reasoning_lang = Language.EN
            decision.output_lang = Language.EN  # 用户英文输入 → 英文输出
            decision.decision_reason = "input_en_efficient"
            chain.append("input_en + ratio>1.2 → en")
            decision.decision_chain = chain
            return decision
        
        # 若用户输入中文且模型中文效率高 → 中文推理
        if signals.detected_input_lang == Language.ZH and model_profile.lang_ratio < 0.85:
            decision.reasoning_lang = Language.ZH
            decision.output_lang = Language.ZH
            decision.decision_reason = "input_zh_efficient"
            chain.append("input_zh + ratio<0.85 → zh")
            decision.decision_chain = chain
            return decision
        
        # ---- 性价比阈值层 ----
        ratio = model_profile.lang_ratio
        if ratio > 1.35:
            # 中文极费Token → 切换英文
            decision.reasoning_lang = Language.EN
            # 输出语言始终跟随用户输入 (除非硬规则强制)
            decision.output_lang = signals.detected_input_lang if signals.detected_input_lang != Language.UNKNOWN else Language.ZH
            decision.decision_reason = f"lang_ratio_high_{ratio:.2f}"
            chain.append(f"ratio {ratio:.2f} > 1.35 → en")
        elif ratio < 0.85:
            # 中文极省Token → 中文推理
            decision.reasoning_lang = Language.ZH
            decision.output_lang = Language.ZH
            decision.decision_reason = f"lang_ratio_low_{ratio:.2f}"
            chain.append(f"ratio {ratio:.2f} < 0.85 → zh")
        else:
            # 0.85 ~ 1.35 区间 → 跟随模型原生语言
            decision.reasoning_lang = model_profile.native_lang
            decision.output_lang = signals.detected_input_lang if signals.detected_input_lang != Language.UNKNOWN else Language.ZH
            decision.decision_reason = f"native_fallback_{model_profile.native_lang.value}"
            chain.append(f"ratio {ratio:.2f} in [0.85,1.35] → native({model_profile.native_lang.value})")
        
        # 如果输出语言是UNKNOWN，默认中文
        if decision.output_lang == Language.UNKNOWN:
            decision.output_lang = Language.ZH
        
        decision.decision_chain = chain
        return decision


# ======================== Phase 2: 推理中转换与调用 ========================

class InputTransformer:
    """输入转换枢纽 (保护标识符 + 翻译)"""
    
    # 占位符保护
    PLACEHOLDER_PATTERNS = [
        (r'`[^`]+`', '__CODE_PLACEHOLDER__'),           # 行内代码
        (r'```[\s\S]+?```', '__BLOCK_PLACEHOLDER__'),   # 代码块
        (r'[a-zA-Z_][a-zA-Z0-9_.-]+(?=\s*[=\(\[\{])', '__IDENT_PLACEHOLDER__'),  # 变量名
    ]
    
    @classmethod
    def _translate(cls, text: str, target: Language) -> str:
        """
        轻量级翻译 (实际调用DeepSeek-V4-Flash或NLLB-200)
        此处为占位, 集成时替换为真实的翻译API
        """
        # 生产环境请替换为:
        # return translation_api.translate(text, target=target.value)
        if target == Language.EN:
            # 模拟: 简单规则 (实际请用真实翻译)
            return text  # 占位
        return text
    
    @classmethod
    def transform(cls, text: str, history: List[Dict], target_lang: Language) -> Tuple[str, List[Dict]]:
        """将自然语言输入转换为标准化推理指令"""
        
        # 1. 保护标识符 (防止翻译破坏代码/API)
        protected_text = text
        placeholder_map = {}
        for pattern, placeholder in cls.PLACEHOLDER_PATTERNS:
            matches = re.findall(pattern, protected_text)
            for i, match in enumerate(matches):
                key = f"{placeholder}_{i}"
                protected_text = protected_text.replace(match, key)
                placeholder_map[key] = match
        
        # 2. 翻译 (如果需要)
        if target_lang != Language.EN:  # 简化: 实际需判断当前语言
            translated = cls._translate(protected_text, target_lang)
        else:
            translated = protected_text
        
        # 3. 还原占位符
        for key, original in placeholder_map.items():
            translated = translated.replace(key, original)
        
        # 4. 历史上下文统一重写
        rewritten_history = []
        for turn in history:
            content = turn.get('content', '')
            if target_lang == Language.EN:
                # 如果历史是中文, 翻译为英文
                # 简化为: 检测中文字符比例
                chinese_ratio = sum(1 for c in content if '\u4e00' <= c <= '\u9fff') / max(len(content), 1)
                if chinese_ratio > 0.3:
                    content = cls._translate(content, Language.EN)
            else:
                # 如果历史是英文, 翻译为中文
                ascii_ratio = sum(1 for c in content if ord(c) < 128) / max(len(content), 1)
                if ascii_ratio > 0.7:
                    content = cls._translate(content, Language.ZH)
            rewritten_history.append({**turn, 'content': content})
        
        return translated, rewritten_history


class SystemPromptInjector:
    """动态System Prompt注入器"""
    
    @classmethod
    def build_prompt(cls, reasoning_lang: Language, output_lang: Language) -> str:
        """根据目标语言生成最强约束的System Prompt"""
        
        if reasoning_lang == Language.EN:
            return f"""You are a helpful AI assistant.
LANGUAGE POLICY (CRITICAL):
1. INTERNAL REASONING: Your chain-of-thought, analysis, and all internal thinking MUST be strictly in ENGLISH.
2. FINAL OUTPUT: Your final response MUST be in {"ENGLISH" if output_lang == Language.EN else "NATURAL CHINESE"}.
3. If ANY character in the reasoning violates this language policy, the response will be discarded.
4. Use <thinking> for reasoning and <answer> for the final output.
    
IMPORTANT: The user will only see the <answer> section. Ensure the final output is fluent and natural."""
        else:
            return f"""你是一个有帮助的AI助手。
语言策略 (最高优先级)：
1. 内部推理：你的思考链、分析和所有内部思考必须严格使用中文。
2. 最终回答：你的最终回答必须使用{"中文" if output_lang == Language.ZH else "英文"}。
3. 如果推理过程中出现任何违反语言策略的字符，回答将被丢弃。
4. 使用<thinking>标签进行推理，使用<answer>标签输出最终回答。
    
重要：用户只会看到<answer>部分的内容。确保最终输出自然流畅。"""


# ======================== Phase 3: 输出后处理与质量保障 ========================

class OutputPostProcessor:
    """输出解析、漂移检测、术语修正"""
    
    # 通用术语表 (英文→中文)
    GLOSSARY = {
        "gradient descent": "梯度下降",
        "backpropagation": "反向传播",
        "attention mechanism": "注意力机制",
        "overfitting": "过拟合",
        "underfitting": "欠拟合",
        "loss function": "损失函数",
        "activation function": "激活函数",
        "regularization": "正则化",
        "embedding": "嵌入向量",
        "semantic": "语义",
        "token": "词元",
        "batch size": "批量大小",
        "learning rate": "学习率",
        "epoch": "轮次",
    }
    
    @classmethod
    def parse_output(cls, raw: str) -> Tuple[str, str]:
        """解析 <thinking> 和 <answer> 标签"""
        reasoning = ""
        answer = raw
        
        # 尝试提取思考内容
        think_pattern = r'<thinking[^>]*>(.*?)</thinking>'
        match = re.search(think_pattern, raw, re.DOTALL | re.IGNORECASE)
        if match:
            reasoning = match.group(1).strip()
        
        # 提取最终回答
        answer_pattern = r'<answer[^>]*>(.*?)</answer>'
        match = re.search(answer_pattern, raw, re.DOTALL | re.IGNORECASE)
        if match:
            answer = match.group(1).strip()
        else:
            # 如果没有answer标签, 去掉thinking部分后作为答案
            answer = re.sub(r'<thinking[^>]*>.*?</thinking>', '', raw, flags=re.DOTALL | re.IGNORECASE).strip()
        
        return reasoning, answer
    
    @classmethod
    def detect_language_drift(cls, text: str, expected_lang: Language) -> bool:
        """检测是否发生语言漂移 (目标语言占比低于阈值)"""
        if not text:
            return True
        
        if expected_lang == Language.ZH:
            chinese_chars = sum(1 for c in text if '\u4e00' <= c <= '\u9fff')
            ratio = chinese_chars / max(len(text), 1)
            # 中文期望输出中, 中文字符低于60%视为漂移
            return ratio < 0.60
        else:
            ascii_chars = sum(1 for c in text if ord(c) < 128)
            ratio = ascii_chars / max(len(text), 1)
            # 英文期望输出中, ASCII字符低于60%视为漂移
            return ratio < 0.60
    
    @classmethod
    def apply_glossary(cls, text: str, direction: str = "en2zh") -> str:
        """术语表替换"""
        if direction == "en2zh":
            for en, zh in cls.GLOSSARY.items():
                # 简单替换 (生产环境建议用spaCy或词边界匹配)
                text = re.sub(r'\b' + re.escape(en) + r'\b', zh, text, flags=re.IGNORECASE)
        return text
    
    @classmethod
    def polish(cls, text: str, target_lang: Language) -> str:
        """
        意译润色 (实际调用轻量模型)
        这里做轻量级处理: 去除直译腔 + 术语标准化
        """
        # 1. 术语标准化
        if target_lang == Language.ZH:
            text = cls.apply_glossary(text, "en2zh")
        
        # 2. 去除多余空格和换行
        text = re.sub(r'\s+', ' ', text).strip()
        
        # 3. 修复明显直译 (占位, 实际需调用润色模型)
        # 生产环境: 调用轻量模型做最终润色, 如 DeepSeek-V4-Flash
        
        return text


# ======================== 网关主控制器 ========================

class InferenceLifecycleGateway:
    """推理语言全生命周期管理网关"""
    
    def __init__(self, model_profiles: Dict[str, ModelProfile]):
        """
        Args:
            model_profiles: 从数据库加载的模型画像, key=model_id
        """
        self.profiles = model_profiles
        
    def process(
        self, 
        raw_input: str, 
        user_id: str, 
        model_id: str,
        history: List[Dict] = None
    ) -> Dict[str, Any]:
        """
        网关主入口: 标准化处理用户输入 -> 调用模型 -> 返回最终输出
        
        Args:
            raw_input: 用户原始自然语言输入
            user_id: 用户标识
            model_id: 目标模型ID
            history: 多轮对话历史 [{"role": "user/assistant", "content": "..."}]
        
        Returns:
            标准化后的结果字典, 包含: request_id, final_output, telemetry
        """
        history = history or []
        start_time = time.time()
        request_id = f"req_{datetime.now().strftime('%Y%m%d%H%M%S')}_{user_id[:8]}"
        
        # ---------- Phase 1: 信号提取 ----------
        phase1_start = time.time()
        signals = SignalExtractor.extract(raw_input, history)
        model_profile = self.profiles.get(model_id)
        if not model_profile:
            raise ValueError(f"Model profile not found: {model_id}")
        
        routing = Router.decide(signals, model_profile)
        phase1_latency = (time.time() - phase1_start) * 1000  # ms
        
        # ---------- Phase 2: 推理中转换 ----------
        phase2_start = time.time()
        
        # 2.1 输入转换
        transformed_input, rewritten_history = InputTransformer.transform(
            raw_input, history, routing.reasoning_lang
        )
        
        # 2.2 System Prompt 注入
        system_prompt = SystemPromptInjector.build_prompt(
            routing.reasoning_lang, routing.output_lang
        )
        
        # 2.3 构造标准化请求
        canonical = CanonicalRequest(
            request_id=request_id,
            user_id=user_id,
            raw_input=raw_input,
            signals=signals,
            routing=routing,
            system_prompt=system_prompt,
            user_prompt=transformed_input,
            history=rewritten_history
        )
        
        # 2.4 调用模型 (模拟)
        # 实际部署替换为真实的API调用
        raw_response = self._call_model(canonical, model_profile)
        phase2_latency = (time.time() - phase2_start) * 1000
        
        # ---------- Phase 3: 输出后处理 ----------
        phase3_start = time.time()
        
        # 3.1 解析标签
        reasoning, final_answer = OutputPostProcessor.parse_output(raw_response)
        
        # 3.2 语言漂移检测
        if OutputPostProcessor.detect_language_drift(final_answer, routing.output_lang):
            # 触发紧急修正: 调用翻译枢纽强制转回目标语言
            final_answer = InputTransformer._translate(final_answer, routing.output_lang)
        
        # 3.3 术语一致性与润色
        final_output = OutputPostProcessor.polish(final_answer, routing.output_lang)
        
        phase3_latency = (time.time() - phase3_start) * 1000
        
        # ---------- 构造返回 ----------
        total_latency = (time.time() - start_time) * 1000
        canonical.raw_response = raw_response
        canonical.final_output = final_output
        canonical.telemetry = {
            "phase1_latency_ms": round(phase1_latency, 2),
            "phase2_latency_ms": round(phase2_latency, 2),
            "phase3_latency_ms": round(phase3_latency, 2),
            "total_latency_ms": round(total_latency, 2),
            "decision_chain": routing.decision_chain,
            "reasoning_lang": routing.reasoning_lang.value,
            "output_lang": routing.output_lang.value,
        }
        
        return {
            "request_id": request_id,
            "final_output": final_output,
            "telemetry": canonical.telemetry,
            "canonical": canonical,  # 完整数据结构, 用于日志
        }
    
    def _call_model(self, canonical: CanonicalRequest, profile: ModelProfile) -> str:
        """
        实际调用模型 (API/本地)
        此处为模拟占位, 集成时替换
        """
        # 构造OpenAI兼容的消息格式
        messages = [
            {"role": "system", "content": canonical.system_prompt}
        ]
        # 加入历史 (已转换)
        for turn in canonical.history:
            messages.append({"role": turn.get("role", "user"), "content": turn.get("content", "")})
        # 加入当前用户输入 (已转换)
        messages.append({"role": "user", "content": canonical.user_prompt})
        
        # 模拟响应 (实际调用 openai.ChatCompletion 或本地 vLLM)
        # 模拟中英混合输出, 测试后处理能力
        if canonical.routing.reasoning_lang == Language.EN:
            sample = "<thinking>Let me analyze this step by step.</thinking>\n<answer>根据分析，答案是42。</answer>"
        else:
            sample = "<thinking>让我们一步步分析这个问题。</thinking>\n<answer>根据分析，答案是42。</answer>"
        return sample
