def parameterize_test_cases(test_cases):
    """参数化测试用例装饰器"""
    def decorator(func):
        @pytest.mark.parametrize("test_input,expected", test_cases)
        def wrapper(test_input, expected):
            return func(test_input, expected)
        return wrapper
    return decorator

# 使用示例

### 2.3 配置验证逻辑重构

#### 2.3.1 创建通用验证器基类
**目标**：统一安全验证逻辑

**方案**：
```python
# scripts/core/security_base.py中添加
from abc import ABC, abstractmethod
from dataclasses import dataclass
from typing import Optional

@dataclass
class ValidationResult:
    """验证结果通用结构"""
    is_valid: bool
    message: str
    data: Optional[Any] = None
    risk_level: str = "LOW"

class BaseValidator(ABC):
    """验证器基类"""
    
    def __init__(self, max_length: int = 1024, allowed_patterns: List[str] = None):
        self.max_length = max_length
        self.allowed_patterns = allowed_patterns or []
        self.dangerous_patterns = self._get_dangerous_patterns()
    
    @abstractmethod
    def validate(self, value: str) -> ValidationResult:
        """验证方法"""
        pass
    
    def _validate_length(self, value: str) -> Optional[str]:
        """长度验证通用方法"""
        if len(value) > self.max_length:
            return f"Value exceeds maximum length {self.max_length}"
        return None
    
    def _validate_patterns(self, value: str) -> Optional[str]:
        """模式验证通用方法"""
        for pattern in self.dangerous_patterns:
            if re.search(pattern, value):
                return f"Value contains dangerous pattern: {pattern}"
        return None
    
    def _get_dangerous_patterns(self) -> List[str]:
        """获取危险模式列表"""
        return [
            r"\.\./",  # 目录遍历
            r"\.\.\\",  # Windows目录遍历
            r"\|",      # 管道符号
            r"&",       # 后台执行
            r";",       # 命令分隔符
            r"`",       # 命令替换
            r"\$\(.*\)", # 命令替换
        ]
```

### 3.2 security.py中的validate_path函数重构

#### 3.2.1 当前问题分析
`validate_path()`函数复杂度16，主要问题：
1. 多个独立的验证步骤串联
2. 每个验证步骤都有相似的返回模式
3. 条件判断嵌套

#### 3.2.2 重构方案
**目标**：将复杂度降低到8以下

**方案**：
```python
# 使用责任链模式重构
class ValidationStep:
    """验证步骤基类"""
    
    def __init__(self, next_step=None):
        self.next_step = next_step
    
    def validate(self, path: str, data: Dict = None) -> ValidationResult:
        """执行验证"""
        result = self._do_validate(path, data or {})
        if not result.is_valid:
            return result
        elif self.next_step:
            return self.next_step.validate(path, data)
        else:
            return ValidationResult(True, "All validations passed", data)
    
    @abstractmethod
    def _do_validate(self, path: str, data: Dict) -> ValidationResult:
        """具体验证逻辑"""
        pass

class EmptyCheck(ValidationStep):
    """空值检查"""
    
    def _do_validate(self, path: str, data: Dict) -> ValidationResult:
        if not path:
            return ValidationResult(False, "Path is empty", risk_level="HIGH")
        data['original_path'] = path
        return ValidationResult(True, "", data)

class LengthCheck(ValidationStep):
    """长度检查"""
    
    def __init__(self, max_length: int, next_step=None):
        super().__init__(next_step)
        self.max_length = max_length
    
    def _do_validate(self, path: str, data: Dict) -> ValidationResult:
        if len(path) > self.max_length:
            return ValidationResult(
                False, 
                f"Path exceeds maximum length {self.max_length}", 
                risk_level="HIGH"
            )
        return ValidationResult(True, "", data)

# 更多验证步骤类...

# 重构后的validate_path函数
def validate_path(self, path: str, allow_create: bool = False) -> ValidationResult:
    """验证路径安全性 - 复杂度从16降低到<8"""
    # 构建责任链
    chain = EmptyCheck(
        LengthCheck(
            self.manager.max_path_length,
            PatternCheck(
                self.DANGEROUS_PATTERNS,
                PathResolutionCheck(
                    self.ALLOWED_PATH_PREFIXES,
                    self.manager.allowed_paths,
                    ExistenceCheck(allow_create)
                )
            )
        )
    )
    
    return chain.validate(path)
```
